# PyTorch 使用 CPU 轮子源，避免在无 GPU 的云端安装体积巨大的 CUDA 版本
--extra-index-url https://download.pytorch.org/whl/cpu

# Web 界面（1.49 起才支持 button / form_submit_button 的 width 参数）
streamlit==1.56.0

# torch 与 torchvision 必须成对：CPU 源上的 torchvision 0.26.0 要求 torch==2.11.0。
# 这里只写基础版本号、刻意不加 +cpu 后缀：
#   - Linux 上 pip 会选中 2.11.0+cpu（本地版本号高于 2.11.0）
#   - macOS 上则正常从 PyPI 取常规轮子；若写死 +cpu，本地 pip 安装会失败
#   （+cpu 轮子只有 Linux / Windows）
torch==2.11.0
torchvision==0.26.0

# 目标检测框架：会自动带入 numpy / matplotlib / pillow / pyyaml /
# scipy / psutil / polars / pandas / ultralytics-thop 等依赖
ultralytics==8.4.39

# opencv 显式钉住：ultralytics 只要求 >=4.6.0，不钉会装到 5.x，
# 与本地验证过的版本跨度较大（cv2 大版本常有 API 变动）
# 注：本地 conda 显示为 4.13.0，PyPI 上的完整版本号是 4.13.0.92
opencv-python==4.13.0.92
